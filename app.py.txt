class Database:
    def __init__(self, db_name="stok_yonetimi.db"):
        self.conn = sqlite3.connect(db_name)
        self.create_tables()

    def create_tables(self):
        cursor = self.conn.cursor()
        cursor.execute('''
        CREATE TABLE IF NOT EXISTS urunler (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            barkod TEXT UNIQUE,
            urun_adi TEXT,
            stok INTEGER,
            fiyat REAL
        )
        ''')
        cursor.execute('''
        CREATE TABLE IF NOT EXISTS ayarlar (
            anahtar TEXT PRIMARY KEY,
            deger TEXT
        )
        ''')
        self.conn.commit()

    def ayar_getir(self, anahtar):
        cursor = self.conn.cursor()
        cursor.execute("SELECT deger FROM ayarlar WHERE anahtar = ?", (anahtar,))
        sonuc = cursor.fetchone()
        return sonuc[0] if sonuc else ""

    def ayar_kaydet(self, anahtar, deger):
        cursor = self.conn.cursor()
        cursor.execute("INSERT OR REPLACE INTO ayarlar (anahtar, deger) VALUES (?, ?)", (anahtar, deger))
        self.conn.commit()

    def urunleri_getir(self):
        cursor = self.conn.cursor()
        cursor.execute("SELECT * FROM urunler")
        return cursor.fetchall()

    def stok_guncelle_db(self, barkod, yeni_stok):
        cursor = self.conn.cursor()
        cursor.execute("UPDATE urunler SET stok = ? WHERE barkod = ?", (yeni_stok, barkod))
        self.conn.commit()

    def urun_ekle_veya_guncelle(self, barkod, urun_adi, stok, fiyat):
        query = '''
        INSERT INTO urunler (barkod, urun_adi, stok, fiyat)
        VALUES (?, ?, ?, ?)
        ON CONFLICT(barkod) DO UPDATE SET
            urun_adi=excluded.urun_adi,
            stok=excluded.stok,
            fiyat=excluded.fiyat
        '''
        self.conn.execute(query, (barkod, urun_adi, stok, fiyat))
        self.conn.commit()

    def load_settings(self):
        self.ty_supp.setText(self.db.ayar_getir("ty_supp"))
        self.ty_key.setText(self.db.ayar_getir("ty_key"))
        self.ty_sec.setText(self.db.ayar_getir("ty_sec"))
        self.hb_merchant.setText(self.db.ayar_getir("hb_merchant"))
        self.hb_token.setText(self.db.ayar_getir("hb_token"))
        self.ptt_user.setText(self.db.ayar_getir("ptt_user"))
        self.ptt_pass.setText(self.db.ayar_getir("ptt_pass"))

    def save_settings_to_db(self):
        self.db.ayar_kaydet("ty_supp", self.ty_supp.text().strip())
        self.db.ayar_kaydet("ty_key", self.ty_key.text().strip())
        self.db.ayar_kaydet("ty_sec", self.ty_sec.text().strip())
        self.db.ayar_kaydet("hb_merchant", self.hb_merchant.text().strip())
        self.db.ayar_kaydet("hb_token", self.hb_token.text().strip())
        self.db.ayar_kaydet("ptt_user", self.ptt_user.text().strip())
        self.db.ayar_kaydet("ptt_pass", self.ptt_pass.text().strip())
        QMessageBox.information(self, "Başarılı", "API ayarları veritabanına başarıyla kaydedildi!")

    def load_data(self):
        urunler = self.db.urunleri_getir()
        self.table.setRowCount(0)
        for row_idx, urun in enumerate(urunler):
            self.table.insertRow(row_idx)
            for col_idx, value in enumerate(urun):
                item = QTableWidgetItem(str(value))
                if col_idx in [3, 4]:  # Stok ve Fiyat sütunlarını ortala
                    item.setTextAlignment(Qt.AlignCenter)
                self.table.setItem(row_idx, col_idx, item)

    def select_row(self, item):
        row = item.row()
        self.txt_barkod.setText(self.table.item(row, 1).text())
        self.txt_ad.setText(self.table.item(row, 2).text())
        self.txt_stok.setText(self.table.item(row, 3).text())
        self.txt_fiyat.setText(self.table.item(row, 4).text())

    def save_and_sync(self):
        barkod = self.txt_barkod.text().strip()
        ad = self.txt_ad.text().strip()
        stok = self.txt_stok.text().strip()
        fiyat = self.txt_fiyat.text().strip()

        if not (barkod and ad and stok and fiyat):
            QMessageBox.warning(self, "Eksik Alan", "Lütfen tüm ürün alanlarını eksiksiz doldurunuz!")
            return

        self.db.urun_ekle_veya_guncelle(barkod, ad, int(stok), float(fiyat))
        self.load_data()
        self.pazaryerlerine_stok_gonder(barkod, int(stok), float(fiyat))
        QMessageBox.information(self, "Başarılı", "Ürün veritabanına kaydedildi ve pazaryerlerine senkronize edildi.")

    def pazaryerlerine_stok_gonder(self, barkod, stok, fiyat):
        ty_supp = self.db.ayar_getir("ty_supp")
        ty_key = self.db.ayar_getir("ty_key")
        ty_sec = self.db.ayar_getir("ty_sec")

        if ty_supp and ty_key and ty_sec:
            try:
                url = f"https://api.trendyol.com/sapigw/suppliers/{ty_supp}/products/price-and-inventory"
                payload = {"items": [{"barcode": barkod, "quantity": stok, "salePrice": fiyat, "listPrice": fiyat}]}
                requests.post(url, data=json.dumps(payload), headers={"Content-Type": "application/json"}, auth=HTTPBasicAuth(ty_key, ty_sec), timeout=5)
            except:
                pass

    def arkaplanda_siparisleri_kontrol_et(self):
        ty_supp = self.db.ayar_getir("ty_supp")
        ty_key = self.db.ayar_getir("ty_key")
        ty_sec = self.db.ayar_getir("ty_sec")

        if not (ty_supp and ty_key and ty_sec):
            return

        try:
            url = f"https://api.trendyol.com/sapigw/suppliers/{ty_supp}/orders?status=Created"
            response = requests.get(url, auth=HTTPBasicAuth(ty_key, ty_sec), timeout=5)
            
            if response.status_code == 200:
                siparisler = response.json().get("content", [])
                for siparis in siparisler:
                    for kalem in siparis.get("lines", []):
                        barkod = kalem.get("barcode")
                        satilan_adet = kalem.get("quantity")
                        
                        cursor = self.db.conn.cursor()
                        cursor.execute("SELECT stok FROM urunler WHERE barkod = ?", (barkod,))
                        kayit = cursor.fetchone()
                        if kayit:
                            mevcut_stok = kayit[0]
                            yeni_stok = max(0, mevcut_stok - satilan_adet)
                            self.db.stok_guncelle_db(barkod, yeni_stok)
                            self.pazaryerlerine_stok_gonder(barkod, yeni_stok, 0.0)
                self.load_data()
        except Exception as e:
            print("Arka plan senkronizasyon hatası:", e)
